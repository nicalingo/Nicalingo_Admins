-- 1. Activar RLS en las tablas de contenido
ALTER TABLE levels ENABLE ROW LEVEL SECURITY;
ALTER TABLE lessons ENABLE ROW LEVEL SECURITY;
ALTER TABLE questions ENABLE ROW LEVEL SECURITY;
ALTER TABLE question_options ENABLE ROW LEVEL SECURITY;

-- ==========================================
-- POLÍTICAS DE LECTURA (Cualquier usuario puede estudiar/ver el contenido)
-- ==========================================
CREATE POLICY "Lectura libre de levels" ON levels FOR SELECT USING (true);
CREATE POLICY "Lectura libre de lessons" ON lessons FOR SELECT USING (true);
CREATE POLICY "Lectura libre de questions" ON questions FOR SELECT USING (true);
CREATE POLICY "Lectura libre de question_options" ON question_options FOR SELECT USING (true);

-- ==========================================
-- POLÍTICAS DE ESCRITURA (INSERT, UPDATE, DELETE solo para editores o admins)
-- ==========================================

-- Para LEVELS
CREATE POLICY "Escritura segura en levels" ON levels 
FOR ALL TO authenticated 
USING (public.get_user_role(auth.uid()) IN ('editor', 'admin'))
WITH CHECK (public.get_user_role(auth.uid()) IN ('editor', 'admin'));

-- Para LESSONS
CREATE POLICY "Escritura segura en lessons" ON lessons 
FOR ALL TO authenticated 
USING (public.get_user_role(auth.uid()) IN ('editor', 'admin'))

-- Para QUESTION_OPTIONS
